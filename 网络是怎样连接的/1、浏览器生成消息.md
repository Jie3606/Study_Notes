#!/usr/bin/expect  
  
# 设置变量  
set timeout 20  
set uart_device /dev/ttyUSB0  # UART设备路径  
set baud_rate 115200          # 波特率  
set embedded_ip 192.168.1.100 # 假设要测试的嵌入式设备的IP地址  
  
# 启动串行通信会话  
spawn screen $uart_device $baud_rate  
  
# 等待嵌入式设备启动完成（如果需要）  
expect "login:" { send "root\r" }  # 假设需要登录，发送用户名  
expect "Password:" { send "password\r" }  # 假设需要密码，发送密码  
  
# 发送网络测试命令并捕获输出  
expect "$ " { send "ifconfig\r" }  # 获取IP配置信息  
expect -re "inet addr:([0-9.]+)\s" { set actual_ip $expect_out(1,string) }  # 解析IP地址  
expect "$ "  
  
# 比较实际IP与期望IP  
if { [string compare $actual_ip $embedded_ip] == 0 } {  
    puts "IP address test passed. Actual IP: $actual_ip"  
} else {  
    puts "IP address test failed. Expected IP: $embedded_ip, Actual IP: $actual_ip"  
    exit 1  
}  
  
# 发送ping命令测试网络连接（假设嵌入式设备支持ping命令）  
expect "$ " { send "ping -c 4 8.8.8.8\r" }  # ping Google的DNS服务器  
expect -re "(\d+) packets transmitted, (\d+) received" {  
    set transmitted $expect_out(1,string)  
    set received $expect_out(2,string)  
}  
expect "$ "  
  
# 检查ping结果  
if { [string compare $transmitted $received] == 0 } {  
    puts "Ping test passed."  
} else {  
    puts "Ping test failed. Transmitted: $transmitted, Received: $received"  
    exit 1  
}  
  
# 退出串行通信会话  
expect "$ " { send "exit\r" }  
expect eof  
  
# 脚本结束  
puts "All network tests completed."  
exit 0

#!/bin/bash

# 串口设备文件，根据你的系统调整
UART_DEVICE="/dev/ttyUSB0"

# 设置串口参数（波特率、字符大小、无硬件流控、无奇偶校验）
stty -F $UART_DEVICE 115200 cs8 -cstopb -parenb ignbrk -icrnl -ixon -opost -onlcr

# 目标设备的IP地址或主机名（此命令将发送到嵌入式设备）
TARGET_HOST="192.168.1.1"

# 日志文件路径
LOG_FILE="uart_communication.log"

# 读取超时时间（秒）
READ_TIMEOUT=5

# 每次读取的数据块大小（字节）
CHUNK_SIZE=64

# 检查日志文件是否存在，不存在则创建
if [ ! -f "$LOG_FILE" ]; then
    touch "$LOG_FILE"
fi

# 初始化变量，存储读取到的完整响应
response=""

# 无限循环，持续发送数据并读取返回值
while true; do
    echo "[$(date)] 等待嵌入式设备启动..." | tee -a "$LOG_FILE"
    
    # 等待设备启动并显示登录提示
    while ! echo "ping" | nc -w1 $UART_DEVICE 2>/dev/null; do
        sleep 1
    done
    
    echo "[$(date)] 嵌入式设备已启动，准备发送命令" | tee -a "$LOG_FILE"
    
    # 通过串口发送数据到嵌入式设备
    echo "ping -c 4 $TARGET_HOST" > $UART_DEVICE
    
    echo "[$(date)] 等待嵌入式设备返回结果..." | tee -a "$LOG_FILE"
    
    # 开始读取数据，设置一个超时读取循环
    end_time=$((SECONDS + READ_TIMEOUT))
    
    while [ $SECONDS -lt $end_time ]; do
        # 读取数据块
        chunk=$(dd if=$UART_DEVICE bs=$CHUNK_SIZE count=1 2>/dev/null)
        
        # 如果有数据返回，则将其拼接到response变量
        if [ -n "$chunk" ]; then
            response+="$chunk"
        else
            # 如果没有读取到数据，等待一小段时间再试
            sleep 0.1
        fi
        
        # 你可以在这里添加逻辑来检查是否已经读取到完整响应（例如检查特定结束符）
        # if [[ "$response" == *"结束符"* ]]; then
        #     break
        # fi
    done
    
    # 将结果输出到日志文件
    if [ -n "$response" ]; then
        echo "[$(date)] 设备返回: $response" | tee -a "$LOG_FILE"
    else
        echo "[$(date)] 读取超时，未收到完整响应" | tee -a "$LOG_FILE"
    fi
    
    # 等待下一个操作
    sleep 5
done
