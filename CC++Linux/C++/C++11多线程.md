

wait()用来等一个东西

如果第二个参数lambda表达式的返回值是false,那么wait()将互斥量解锁，并阻塞到本行，

阻塞到什么时候嘞?阻塞到其他函数来调用notify_one()成员函数为，之后wait()不断的尝试获取互斥量锁，如果获取不到那么就会卡在wait()这里等待获取，如果获取到锁wait()就继续执行，如果wait()没有第二个参数，那么就默认lambda返回false效果一样。



- 如果wait()有第二个参数（lambda），就判断这个表达式，如果为false继续解锁休眠，等待在次被notify唤醒，继续阻塞，为true wait 返回
- 没有第二个参数就返回

# abc三个线程交替打印0 1 2 3 4 5 … … (会产生死锁)

```c++
#include<iostream>
#include<mutex>
#include<thread>
#include<condition_variable>
using namespace std;

std::mutex g_mtx;
std::condition_variable g_cv;
int num = 0;
const int n = 10;

void funa()
{
	std::unique_lock<std::mutex> lock(g_mtx);
	while (num < n)
	{
		while (num % 3 == 0) // 0 3 6 9
		{
			cout << "tha: " << num << endl;
			num += 1;
			g_cv.notify_all();
		}
		g_cv.wait(lock);
	}
}
void funb()
{
	std::unique_lock<std::mutex> lock(g_mtx);
	while (num < n)
	{
		while (num % 3 == 1)// 1 4 7
		{
			cout << "thb: " << num << endl;
			num += 1;
			g_cv.notify_all();
		}
		g_cv.wait(lock);
	}
}
void func()
{
	std::unique_lock<std::mutex> lock(g_mtx);
	while (num < n)
	{
		while (num % 3 == 2)//2 5 8
		{
			cout << "thc: " << num << endl;
			num += 1;
			g_cv.notify_all();
		}
		g_cv.wait(lock);
	}
}

int main()
{
	std::thread tha(funa);
	std::thread thb(funb);
	std::thread thc(func);

	tha.join();
	thb.join();
	thc.join();

	return 0;
}
```

# 生产者消费者

```c++
#include<iostream>
#include<mutex>
#include<thread>
#include<condition_variable>
using namespace std;

std::mutex g_mtx;
std::condition_variable g_cv;

bool rs ;
const int n = 4;
int num;

void producer()
{
	std::unique_lock<std::mutex> lock(g_mtx);
	while (num < n)
	{
		while (rs)
		{
			g_cv.wait(lock);
		}
		cout << "produce       " << endl;
		rs = true;
		num++;
		g_cv.notify_all();
	}
}

void consumer()
{
	int num = 0;
	std::unique_lock<std::mutex> lock(g_mtx);
	while (num < n)
	{
		while (!rs)
		{
			g_cv.wait(lock);
		}
		cout << "         consumer" << endl;
		rs = false;
		num++;
		g_cv.notify_all();
		//g_cv.notify_one();
	}
}

int main()
{
	std::thread tha(producer);
	std::thread thb(consumer);
	
	tha.join();
	thb.join();

	return 0;
}
```

# 生产者-消费者队列

```c++
class SyncQueue
{
	std::deque<int> que;
	int maxsize;
	std::mutex mtx;
	std::condition_variable cv;
public:
	SyncQueue(int size = 8) :maxsize(size) {}
	~SyncQueue() {}
	bool isFull()
	{
		return que.size() == maxsize;
	}
	bool isEmpty()const
	{
		return que.empty();
	}
	void put(int data)
	{
		std::unique_lock<std::mutex> lock(mtx);
		while (isFull())
		{
			cv.wait(lock);
		}
		que.push_back(data);
		cv.notify_all();
    }
	int get()
	{
		std::unique_lock<std::mutex> lock(mtx);
		while (isEmpty())
		{
			cv.wait(lock);
		}
	    int tmp = que.front();
		que.pop_front();
		cv.notify_all();
		return tmp;
	}	
};
```
